-- R14 | READ | Availability check: is the space free for the requested slot? No overlapping active booking and no overlapping blocked slot
-- EXPLAIN (ANALYZE, BUFFERS), 3 runs on the seed data (PostgreSQL 18)

-- Query:
SELECT NOT EXISTS (
         SELECT 1 FROM bookings b
         WHERE b.space_id = 1
           AND b.status IN ('PENDING', 'CONFIRMED')
           AND tstzrange(b.starts_at, b.ends_at, '[)')
               && tstzrange(date_trunc('day', now()) + interval '1 day 10 hours',
                            date_trunc('day', now()) + interval '1 day 12 hours', '[)')
       )
       AND NOT EXISTS (
         SELECT 1 FROM blocked_slots bs
         WHERE bs.space_id = 1
           AND tstzrange(bs.starts_at, bs.ends_at, '[)')
               && tstzrange(date_trunc('day', now()) + interval '1 day 10 hours',
                            date_trunc('day', now()) + interval '1 day 12 hours', '[)')
       ) AS is_available;
